
## 一、现有资金表分析

#### 新增表1：tb_bankruptcy_expense（破产费用管理表）

**核心功能**：记录破产费用发生、支付、管理情况，确保破产费用合规合理

**字段设计**：
```sql
CREATE TABLE `tb_bankruptcy_expense` (

  `unpaid_amount` decimal(18,2) DEFAULT NULL COMMENT '未支付金额',

  -- 费用依据

  -- 费用用途

  -- 审批信息
  `approval_status` varchar(20) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci DEFAULT 'PENDING' COMMENT '审批状态: PENDING-待审批, APPROVED-已通过, REJECTED-已驳回',

  `approval_date` datetime DEFAULT NULL COMMENT '审批日期',

  `payment_method` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '支付方式: BANK_TRANSFER-银行转账, CASH-现金, CHECK-支票, OTHER-其他',
  `payment_date` datetime DEFAULT NULL COMMENT '支付日期',

  `payee_account` varchar(100) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '收款人账号',
  `payee_bank` varchar(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '收款人开户行',

  KEY `idx_case_id` (`case_id`),
  KEY `idx_expense_type` (`expense_type`),
  KEY `idx_approval_status` (`approval_status`),
  KEY `idx_payment_status` (`payment_status`),
  KEY `idx_approver_id` (`approver_id`),
  KEY `idx_payment_date` (`payment_date`),
  KEY `idx_status` (`status`),
```
